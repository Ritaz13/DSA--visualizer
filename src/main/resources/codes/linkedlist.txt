// LINKED LIST DATA STRUCTURE
class Node {
    int data;
    Node next;
    Node prev;
    
    Node(int data) {
        this.data = data;
    }
}

class LinkedList {
    private Node head;
    private boolean isDoubly;
    
    // Insert at beginning
    public void insertAtBegin(int data) {
        Node newNode = new Node(data);
        newNode.next = head;
        if (head != null && isDoubly) {
            head.prev = newNode;
        }
        head = newNode;
    }
    
    // Insert at end
    public void insertAtEnd(int data) {
        Node newNode = new Node(data);
        if (head == null) {
            head = newNode;
            return;
        }
        Node current = head;
        while (current.next != null) {
            current = current.next;
        }
        current.next = newNode;
        if (isDoubly) newNode.prev = current;
    }
    
    // Delete node
    public void delete(int data) {
        if (head == null) return;
        if (head.data == data) {
            head = head.next;
            return;
        }
        Node current = head;
        while (current.next != null) {
            if (current.next.data == data) {
                current.next = current.next.next;
                return;
            }
            current = current.next;
        }
    }
    
    // Search for element
    public boolean search(int data) {
        Node current = head;
        while (current != null) {
            if (current.data == data) return true;
            current = current.next;
        }
        return false;
    }
    
    // Reverse linked list
    public void reverse() {
        Node prev = null, current = head;
        while (current != null) {
            Node next = current.next;
            current.next = prev;
            prev = current;
            current = next;
        }
        head = prev;
    }
}
