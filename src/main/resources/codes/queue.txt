// QUEUE DATA STRUCTURE
class Queue {
    private int[] queue;
    private int front, rear;
    private int capacity;
    
    public Queue(int capacity) {
        this.capacity = capacity;
        queue = new int[capacity];
        front = 0;
        rear = -1;
    }
    
    // Enqueue element
    public void enqueue(int value) {
        if (rear == capacity - 1) {
            System.out.println("Queue is Full");
            return;
        }
        queue[++rear] = value;
    }
    
    // Dequeue element
    public int dequeue() {
        if (front > rear) {
            System.out.println("Queue is Empty");
            return -1;
        }
        return queue[front++];
    }
    
    // Get front element
    public int getFront() {
        if (front > rear) return -1;
        return queue[front];
    }
    
    // Get rear element
    public int getRear() {
        if (rear == -1) return -1;
        return queue[rear];
    }
    
    // Check if empty
    public boolean isEmpty() {
        return front > rear;
    }
    
    // Circular Queue: reuses space
    public void enqueueCircular(int value) {
        rear = (rear + 1) % capacity;
        queue[rear] = value;
    }
}
